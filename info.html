Commands in VMUL and what's translated in HTML:

# -> h1
## -> h2
* -> i
** -> b
! -> title
% -> var
_ -> u
- -> s
$\n -> br
$FULLBAR -> meter value = "1"
$SEMIBAR -> meter value = "0.5"
$ZEROBAR -> meter value = "0"
$HASTAG -> #
$DOUBLEHASTAG -> ##
$EXCLAMATIONMARK -> !
$ASTERISK -> *
$DOUBLEASTERISK -> **
$UNDERLINE -> _
$HYPHEN -> -
